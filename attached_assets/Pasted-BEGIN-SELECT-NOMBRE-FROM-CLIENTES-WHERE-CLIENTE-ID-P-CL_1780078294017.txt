BEGIN
	SELECT NOMBRE FROM CLIENTES WHERE CLIENTE_ID=:P_CLIENTE_ID INTO
		NOMBRE_CLIENTE;
	CLAVE_CLIENTE = NULL;
	SELECT CLAVE_CLIENTE FROM CLAVES_CLIENTES WHERE CLIENTE_ID=:P_CLIENTE_ID AND
		ROL_CLAVE_CLI_ID=2 INTO CLAVE_CLIENTE;
	FECHA_SALDO_INI = P_FECHA_INICIAL-1;
	/* Calcular saldo inicial */
    EXECUTE PROCEDURE CALC_SALDO_CLIENTE P_CLIENTE_ID, FECHA_SALDO_INI, 'S'
		RETURNING_VALUES SALDO_INICIAL, SALDO_XACR;
    TUVO_MOVTOS = 'N';
	FOR SELECT DOCTO_CC_ID, MOVTO_ESPECIAL, SUM(IMPORTE_CARGO), SUM(IMPORTE_CREDITO), 
               SUM(SALDO_FIN)
        FROM AUXILIAR_CLIENTE(:P_CLIENTE_ID, :P_FECHA_INICIAL, :P_FECHA_FINAL,
                              :SALDO_INICIAL, 'S')
        GROUP BY DOCTO_CC_ID, MOVTO_ESPECIAL 
        INTO DOCTO_CC_ID, MOVTO_ESPECIAL, CARGO, CREDITO, SALDO_FINAL       
    DO
    BEGIN
		/* Obtenemos los datos necesarios del concepto */
		SELECT A.FECHA, A.FOLIO, B.NOMBRE_ABREV, B.TIPO, B.ID_INTERNO,A.DESCRIPCION,
			A.NATURALEZA_CONCEPTO,A.COND_PAGO_ID,A.SUCURSAL_ID
			FROM DOCTOS_CC A JOIN CONCEPTOS_CC B ON (A.CONCEPTO_CC_ID = B.CONCEPTO_CC_ID)
			WHERE A.DOCTO_CC_ID = :DOCTO_CC_ID
			INTO FECHA, FOLIO, NOMBRE_CONCEPTO, TIPO_CONCEPTO, ID_INTERNO,DESCRIPCION,
			NATURALEZA_CONCEPTO,COND_PAGO_ID,SUCURSAL_ID;
		SELECT NOMBRE FROM SUCURSALES WHERE SUCURSAL_ID=:SUCURSAL_ID INTO SUCURSAL_REP;
		/* Venta de mostrador, regresar descripcion en folio */
		
		IF (ID_INTERNO='f') THEN
			FOLIO= DESCRIPCION;
		FECHA_VENCIMIENTO = NULL;
		/* Calcular vencimiento si es cargo */
		IF (NATURALEZA_CONCEPTO='C') THEN
		BEGIN
			FOR SELECT DIAS_PLAZO FROM PLAZOS_COND_PAG WHERE COND_PAGO_ID=:COND_PAGO_ID
				INTO DIAS_PLAZO
			DO
			BEGIN
				FECHA_VENCIMIENTO = :FECHA + :DIAS_PLAZO;
			END
		END	
		IF ((TIPO_CONCEPTO <> 'T' AND (ID_INTERNO IS NULL OR ID_INTERNO <> 'L')) OR
         (MOVTO_ESPECIAL = 'S')) THEN
		BEGIN
			/* Se trata de aplicacion de anticipo */
			IF ((ID_INTERNO = 'C') OR (TIPO_CONCEPTO = 'M')) THEN
			BEGIN
				CARGO = 0.0;          
				SELECT B.FOLIO FROM GET_ANTI_FROM_DOCTO_CC(:DOCTO_CC_ID) A JOIN ANTICIPOS_CC B
					ON (A.ANTICIPO_CC_ID = B.ANTICIPO_CC_ID) INTO FOLIO;
			END
        
			/* Indicar en el nombre del concepto cuando se trata de un
			cobro de anticipo para distinguirlo de los cobros normales.*/
			IF (TIPO_CONCEPTO = 'P') THEN
				IF (EXISTS (SELECT ANTICIPO_CC_ID FROM USOS_ANTICIPOS_CC WHERE DOCTO_CC_ID = :DOCTO_CC_ID)) THEN
					NOMBRE_CONCEPTO = 'Cobro de anticipo';
			/* Indicar en el nombre del concepto cuando se trata de un 
			descuento por pronto pago */
			IF (MOVTO_ESPECIAL = 'S') THEN
				NOMBRE_CONCEPTO= 'Descuento por pronto pago';
			TUVO_MOVTOS = 'S';
			SUSPEND;
		END
    END /* for de movtos del auxiliar */
    /* El cliente se regresa si no tuvo movtos pero tiene saldo */
    IF ((TUVO_MOVTOS = 'N') AND (SALDO_INICIAL > 0)) THEN
    BEGIN
		DOCTO_CC_ID = NULL;
		SUSPEND;
    END /* if */
END
