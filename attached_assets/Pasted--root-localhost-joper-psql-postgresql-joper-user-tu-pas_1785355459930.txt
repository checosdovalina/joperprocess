SELECT s.id AS ship_id, s.status, s.created_at
FROM shipments s
JOIN orders o ON o.id = s.order_id
JOIN quotations q ON q.id = o.quotation_id
WHERE q.folio = 'MEX-508003';

SELECT r.shipment_id, r.quotation_item_id, r.quantity_released, r.created_at
FROM order_releases r
JOIN orders o ON o.id = r.order_id
JOIN quotations q ON q.id = o.quotation_id
WHERE q.folio = 'MEX-508003'
ORDER BY r.created_at;
