BEGIN
	NO_REGISTRO = 0;
	GSDO_INICIAL=0;
	GSDO_FINAL = 0;
	IF (P_CLIENTE='T') THEN
	BEGIN
		FOR SELECT CLIENTE_ID FROM CLIENTES ORDER BY NOMBRE
			INTO CLIENTE_ID
		DO
		BEGIN
			TIPO='D';
			DESCRIPCION='';		
			SALDO_FINAL_CLIENTE =0;
			SELECT SUM(IMPORTE_CARGO),SUM(IMPORTE_CREDITO)
				FROM AUXILIAR_CLIENTE(:CLIENTE_ID,:P_FECHA_INICIAL,:P_FECHA_FINAL,0,'S')
				INTO TOT_CARGOS,TOT_CREDITOS;
			IF (TOT_CARGOS IS NULL) THEN
				TOT_CARGOS=0;
			IF (TOT_CREDITOS IS NULL) THEN
				TOT_CREDITOS = 0;
			NO_REG_CTE=0;
			FOR SELECT DOCTO_CC_ID,CLAVE_CLIENTE,NOMBRE_CLIENTE,NOMBRE_CONCEPTO,FOLIO,FECHA,
				FECHA_VENCIMIENTO,SALDO_INICIAL,SALDO_FINAL,CARGO,CREDITO,SUCURSAL_REP
				FROM ZAP_AUXCTED(:P_FECHA_INICIAL,:P_FECHA_FINAL,:CLIENTE_ID)
				ORDER BY FECHA,DOCTO_CC_ID
            	INTO DOCTO_CC_ID,CLAVE_CLIENTE,NOMBRE_CLIENTE,NOMBRE_CONCEPTO,FOLIO,FECHA,
				FECHA_VENCIMIENTO,SALDO_INICIAL,SALDO_FINAL,CARGO,CREDITO,SUCURSAL_REP
			
			DO
			BEGIN
				NO_REG_CTE = NO_REG_CTE+1;
				IF (NO_REG_CTE=1) THEN
				BEGIN
					GSDO_INICIAL = GSDO_INICIAL + SALDO_INICIAL;
					SALDO_FINAL_CLIENTE = SALDO_INICIAL + TOT_CARGOS - TOT_CREDITOS;
					GSDO_FINAL = GSDO_FINAL + SALDO_FINAL_CLIENTE;
				END
				ELSE
				BEGIN
					GSDO_INICIAL=0;
					GSDO_FINAL = 0;
				END
				NO_REGISTRO= NO_REGISTRO+1;
				SUSPEND;
			END			
			TIPO='T';
			
			/* Detalle de totales */
			GSDO_INICIAL=0;
			GSDO_FINAL = 0;

			SALDO_FINAL=NULL;
			FECHA=NULL;
			FECHA_VENCIMIENTO=NULL;
			FOLIO=NULL;
			SUCURSAL_REP=NULL;
			NOMBRE_CONCEPTO=NULL;
			ENTRA='N';
			FOR SELECT NOMBRE_CONCEP,SUM(IMPORTE_CARGO),SUM(IMPORTE_CREDITO)
				FROM AUXILIAR_CLIENTE(:CLIENTE_ID,:P_FECHA_INICIAL,:P_FECHA_FINAL,0,'S')
				GROUP BY NOMBRE_CONCEP
				INTO DESCRIPCION,CARGO,CREDITO
			DO
			BEGIN
				SELECT NATURALEZA FROM CONCEPTOS_CC WHERE NOMBRE_ABREV=:DESCRIPCION INTO NATURALEZA;
				ENTRA='S';
				SUSPEND;
			END
			IF (ENTRA='S') THEN
			BEGIN
				DESCRIPCION='CONCEPTO';
				NATURALEZA='A';
				CARGO=NULL;
				CREDITO=NULL;
				SUSPEND;
			END
		END
		CLIENTE_ID=0;
		SALDO_INICIAL=NULL;
		SALDO_FINAL=NULL;
		SALDO_FINAL_CLIENTE=NULL;
		TOT_CARGOS = 0;
		TOT_CREDITOS =0;
		CLAVE_CLIENTE = NULL;
		NOMBRE_CLIENTE = 'ZZZ';
		NOMBRE_CONCEPTO=NULL;
		TIPO = 'Z';
		NO_REGISTRO =NO_REGISTRO+1;
		/* DETALLE DE CONCEPTOS */
		ENTRA = 'N';
		FOR SELECT SUM(CARGO),SUM(CREDITO),NOMBRE_CONCEPTO FROM ZAP_AUXCTET (:P_FECHA_INICIAL,:P_FECHA_FINAL)
			GROUP BY NOMBRE_CONCEPTO
			INTO CARGO,CREDITO,DESCRIPCION
		DO
		BEGIN
			SELECT NATURALEZA FROM CONCEPTOS_CC WHERE NOMBRE_ABREV=:DESCRIPCION INTO NATURALEZA;
			ENTRA='S';
			SUSPEND;			
		END	
		IF (ENTRA='S') THEN
		BEGIN		
			DESCRIPCION='CONCEPTO';
			NATURALEZA='A';
			CARGO=NULL;
			CREDITO=NULL;
			SUSPEND;
		END
	END
	ELSE
	BEGIN
		NO_REG_CTE=0;
		SALDO_FINAL_CLIENTE =0;
		DESCRIPCION='';
		SELECT CLIENTE_ID FROM CLIENTES WHERE UPPER(NOMBRE)=UPPER(:P_CLIENTE) INTO CLIENTE_ID;
		TIPO='D';
		SELECT SUM(IMPORTE_CARGO),SUM(IMPORTE_CREDITO)
			FROM AUXILIAR_CLIENTE(:CLIENTE_ID,:P_FECHA_INICIAL,:P_FECHA_FINAL,0,'S')
			INTO TOT_CARGOS,TOT_CREDITOS;
		IF (TOT_CARGOS IS NULL) THEN
			TOT_CARGOS=0;
		IF (TOT_CREDITOS IS NULL) THEN
			TOT_CREDITOS = 0;

		FOR SELECT DOCTO_CC_ID,CLAVE_CLIENTE,NOMBRE_CLIENTE,NOMBRE_CONCEPTO,FOLIO,FECHA,
			FECHA_VENCIMIENTO,SALDO_INICIAL,SALDO_FINAL,CARGO,CREDITO,SUCURSAL_REP
			FROM ZAP_AUXCTED(:P_FECHA_INICIAL,:P_FECHA_FINAL,:CLIENTE_ID)
			ORDER BY FECHA,DOCTO_CC_ID
			INTO DOCTO_CC_ID,CLAVE_CLIENTE,NOMBRE_CLIENTE,NOMBRE_CONCEPTO,FOLIO,FECHA,
			FECHA_VENCIMIENTO,SALDO_INICIAL,SALDO_FINAL,CARGO,CREDITO,SUCURSAL_REP

		DO
		BEGIN
			NO_REG_CTE = NO_REG_CTE+1;
			IF (NO_REG_CTE=1) THEN
			BEGIN
				GSDO_INICIAL = GSDO_INICIAL + SALDO_INICIAL;
				SALDO_FINAL_CLIENTE = SALDO_INICIAL + TOT_CARGOS - TOT_CREDITOS;
				GSDO_FINAL = GSDO_FINAL + SALDO_FINAL_CLIENTE;
			END
			ELSE
			BEGIN
				TOT_CARGOS =0;
				TOT_CREDITOS =0;
				SALDO_FINAL_CLIENTE = 0;
				GSDO_INICIAL=0;
				GSDO_FINAL = 0;
			END
			NO_REGISTRO= NO_REGISTRO+1;
			SUSPEND;
		END						
		TIPO='T';
		GSDO_INICIAL=0;
		GSDO_FINAL = 0;

		/* Detalle de totales */
		SALDO_FINAL_CLIENTE = 0;
		SALDO_INICIAL=NULL;
		SALDO_FINAL=NULL;
		FECHA=NULL;
		FECHA_VENCIMIENTO=NULL;
		FOLIO=NULL;
		NO_REGISTRO = NO_REGISTRO +1;
		SUCURSAL_REP=NULL;
		NOMBRE_CONCEPTO=NULL;
		NOMBRE_CLIENTE = 'ZZZ';
		ENTRA='N';
		FOR SELECT NOMBRE_CONCEP,SUM(IMPORTE_CARGO),SUM(IMPORTE_CREDITO)
			FROM AUXILIAR_CLIENTE(:CLIENTE_ID,:P_FECHA_INICIAL,:P_FECHA_FINAL,0,'S')
			GROUP BY NOMBRE_CONCEP
			INTO DESCRIPCION,CARGO,CREDITO
		DO
		BEGIN
			SELECT NATURALEZA FROM CONCEPTOS_CC WHERE NOMBRE_ABREV=:DESCRIPCION INTO NATURALEZA;
			ENTRA='S';
			SUSPEND;
		END
		IF (ENTRA='S') THEN
		BEGIN
			DESCRIPCION='CONCEPTO';
			NATURALEZA='A';
			CARGO=NULL;
			CREDITO=NULL;
			SUSPEND;
		END
	END
END
