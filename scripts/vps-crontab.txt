# INSTRUCCIONES DE INSTALACIÓN:
#
# 1. Abre el editor de crontab:
#      crontab -e
#
# 2. Agrega las siguientes líneas (ajusta la ruta del proyecto):
# ================================================================

# --- Variables de entorno (ajusta según tu servidor) ---
DATABASE_URL=postgres://usuario:password@localhost:5432/nexxo

# --- Backup diario a las 2:00 AM ---
0 2 * * * /ruta/al/proyecto/scripts/vps-backup.sh >> /var/log/nexxo-backup.log 2>&1

# --- Backup adicional a las 2:00 PM (opcional, recomendado) ---
0 14 * * * /ruta/al/proyecto/scripts/vps-backup.sh >> /var/log/nexxo-backup.log 2>&1

# ================================================================

#
